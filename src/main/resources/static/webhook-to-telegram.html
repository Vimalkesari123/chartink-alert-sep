Common Integration Examples

1. Python Script (Machine Learning / Cron Job / Web Scraping)

import requests

url = "https://notifyu.me/webhook?uid=YOUR_UID&key=YOUR_KEY"
data = {
    "job": "Nightly Database Backup",
    "status": "Completed",
    "duration": "18s",
    "size": "54MB"
}
requests.post(url, json=data)
